const params = new URLSearchParams(window.location.search); const designName = params.get('design'); const basePrice = parseFloat(params.get('price')); document.getElementById('designName').textContent = designName; document.getElementById('designImage').src = `images/${designName.replace(/\s/g, '')}.png`; document.getElementById('designImage').alt = designName; // Price update logic const lengthEl = document.getElementById('length'); const totalEl = document.querySelector('.total'); function updateTotal() { let price = basePrice; const length = lengthEl.value; if (length === "XXL") price += 1; if (length === "XXXL") price += 2; totalEl.textContent = `Total: $${price}`; } lengthEl.addEventListener('change', updateTotal); updateTotal();
